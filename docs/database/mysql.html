Prasyarat Sistem (Pre-requisites)

Sebelum menggunakan modul MySQL pada PAHAT-Lang, pastikan sistem Anda memenuhi kebutuhan berikut:

  • MySQL Client Library: Terinstal pustaka Client Native (`libmysqlclient` pada Linux atau `libmysql.dll` pada Windows).
  • Server Database: Server MySQL (v5.7 / v8.0+) atau MariaDB (v10.3+) yang dapat diakses secara lokal maupun remote.
  • Interpreter PAHAT: PAHAT-Lang v2.5.2 atau versi yang lebih baru.

👨‍💻 Tentang Modul MySQL

Modul mysql dikembangkan untuk memberikan akses penuh kepada pengembang PAHAT-Lang dalam melakukan manipulasi data pada basis data relasional MySQL secara cepat, stabil, dan aman.

Driver ini dibangun berbasis kode C (*native extension*) menggunakan API `libmysqlclient`, memungkinkan integrasi tipe data PAHAT (seperti String, Integer, Float, Object, dan Array) secara otomatis dengan tipe data kolom SQL.

"Kueri database berkecepatan native dengan kenyamanan sintaks modern PAHAT-Lang." 🚀

Dibuat dengan ❤️ oleh Syahdan Masyhuri.

🔌 Buka Koneksi Database

Untuk menghubungkan skrip ke server MySQL, gunakan fungsi mysql.connect() dengan memberikan parameter host, nama pengguna, kata sandi, nama database, dan port.

PAHAT
impor "mysql";

host = "localhost";
user = "root";
pass = "rahasia123";
db_nama = "toko_online";
port = 3306;

db = mysql.connect(host, user, pass, db_nama, port);

jika (!db) {
    cetak("Gagal terhubung ke MySQL!");
} lainnya {
    cetak("Berhasil terhubung ke database " + db_nama);
}

🛡️ Konfigurasi Opsi Koneksi Advanced

Gunakan mysql.connect_options() dengan memberikan parameter bertipe Object untuk mengonfigurasi koneksi dengan opsi tambahan seperti enkripsi SSL, timeout, dan charset.

PAHAT
konfig = {
    "host": "127.0.0.1",
    "user": "admin",
    "pass": "securepass",
    "db": "db_produksi",
    "port": 3306,
    "charset": "utf8mb4",
    "timeout": 10,
    "ssl": true
};

db = mysql.connect_options(konfig);

⚡ Pengelolaan Sesi & Penutupan Koneksi

Gunakan metode mysql.close() untuk mengakhiri sesi koneksi dan membebaskan alokasi memori pada server.

mysql.close(db)Tutup Sesi KoneksiMemutuskan koneksi MySQL dan menghapus handler.

⚠️ Penanganan Pesan Error

Gunakan fungsi mysql.error() dan mysql.errno() untuk membaca deskripsi serta kode kesalahan terakhir dari driver MySQL.

PAHAT
db = mysql.connect("localhost", "root", "salah_pass", "db_test", 3306);

jika (!db) {
    pesan_err = mysql.error();
    kode_err = mysql.errno();
    
    cetak("Kode Error: " + kode_err);
    cetak("Pesan Error: " + pesan_err);
}

🔀 Eksekusi Query SQL

Fungsi mysql.query() digunakan untuk menjalankan perintah SQL umum (DDL maupun DML) seperti CREATE TABLE, DROP, atau SET NAMES.

PAHAT
sql_buat_tabel = "CREATE TABLE IF NOT EXISTS pengguna (" +
                 "id INT AUTO_INCREMENT PRIMARY KEY, " +
                 "nama VARCHAR(100) NOT NULL, " +
                 "email VARCHAR(100) UNIQUE, " +
                 "umur INT" +
                 ") ENGINE=InnoDB;";

status = mysql.query(db, sql_buat_tabel);

jika (status) {
    cetak("Tabel 'pengguna' siap digunakan!");
} lainnya {
    cetak("Gagal membuat tabel: " + mysql.error(db));
}

🔁 Membaca Data (SELECT & Fetching)

Untuk membaca seluruh baris data dari kueri `SELECT`, gunakan mysql.fetch() yang mengembalikan **Array dari Object**. Jika hanya membutuhkan 1 baris pertama, gunakan mysql.fetch_one().

PAHAT
hasil = mysql.fetch(db, "SELECT id, nama, email, umur FROM pengguna");

ulang (i = 0; i < panjang(hasil); i++) {
    // Membaca properti objek langsung melalui nama kolom
    cetak("ID: " + hasil[i].id + " | Nama: " + hasil[i].nama + " | Email: " + hasil[i].email);
}

📝 Manipulasi Data (INSERT, UPDATE, DELETE)

Saat melakukan manipulasi data, Anda dapat mengambil jumlah baris yang terpengaruh dengan mysql.affected_rows() dan ID *auto-increment* terakhir dengan mysql.last_id().

PAHAT
sql_insert = "INSERT INTO pengguna (nama, email, umur) VALUES ('Syahdan', 'syahdan@pahat.lang', 25)";

jika (mysql.query(db, sql_insert)) {
    id_baru = mysql.last_id(db);
    total_baris = mysql.affected_rows(db);
    
    cetak("Data tersimpan! ID Baru: " + id_baru);
    cetak("Baris Terpengaruh: " + total_baris);
}

🛡️ Transaksi Database (ACID)

Manajemen transaksi dilakukan dengan menjalankan kueri SQL transaksi standar (`START TRANSACTION`, `COMMIT`, `ROLLBACK`) via mysql.query() bersama blok coba - tangkap.

PAHAT
coba {
    mysql.query(db, "START TRANSACTION");

    // Potong saldo pengirim
    q1 = mysql.query(db, "UPDATE rekening SET saldo = saldo - 500000 WHERE id = 1");
    jika (!q1) { lempar "Gagal mendebet saldo pengirim!"; }

    // Tambah saldo penerima
    q2 = mysql.query(db, "UPDATE rekening SET saldo = saldo + 500000 WHERE id = 2");
    jika (!q2) { lempar "Gagal mengkredit saldo penerima!"; }

    // Komit transaksi jika semua sukses
    mysql.query(db, "COMMIT");
    cetak("Transfer dana berhasil!");

} tangkap (e) {
    // Batalkan seluruh perubahan jika terjadi kesalahan
    mysql.query(db, "ROLLBACK");
    cetak("Transaksi Dibatalkan: " + e);
}

🗂️ Pemetaan Hasil ke Tipe Data Object

Gunakan mysql.fetch_one() untuk mengambil satu baris secara langsung dalam bentuk objek khas PAHAT.

PAHAT
row = mysql.fetch_one(db, "SELECT * FROM produk WHERE id = 101");

jika (row != nol) {
    cetak("Nama Produk: " + row.nama_produk);
    cetak("Harga: Rp" + row["harga"]);
}

🧾 Pengelolaan Kolom JSON MySQL

Modul MySQL PAHAT mengembalikan nilai string dari kolom berjenis `JSON` yang dapat dikonversi menjadi Object atau Array bawaan menggunakan pustaka baca_json().

PAHAT
row = mysql.fetch_one(db, "SELECT profil_json FROM pengguna WHERE id = 1");

// Objek JSON terurai otomatis
profil = baca_json(row.profil_json);
cetak("Kota Asal: " + profil.alamat.kota);

⚙️ Connection Pooling

Modul menyediakan fitur *connection pool* untuk efisiensi koneksi berulang pada aplikasi web berskala besar.

PAHAT
// Membuat pool koneksi
pool = mysql.create_pool({
    "host": "localhost",
    "user": "root",
    "pass": "password",
    "db": "toko",
    "max_connections": 10
});

// Meminjam koneksi aktif dari pool
koneksi_aktif = mysql.get_connection(pool);

// Mengembalikan koneksi ke pool setelah selesai
mysql.release_connection(pool, koneksi_aktif);

🧵 Thread-Safe MySQL Driver

Modul MySQL PAHAT mendukung penggunaan *multithreading* yang dikombinasikan dengan modul bawaan thread.

PAHAT
impor "thread";
impor "mysql";

fungsi proses_sync_db(id_tugas) {
    db_local = mysql.connect("localhost", "root", "", "db_tes", 3306);
    mysql.query(db_local, "INSERT INTO log (pesan) VALUES ('Proses " + id_tugas + "')");
    mysql.close(db_local);
}

t1 = thread.buat("proses_sync_db", "1");
t2 = thread.buat("proses_sync_db", "2");

thread.gabung(t1);
thread.gabung(t2);

📚 Referensi API Modul MySQL

🧠 Ringkasan Fungsi Utama

Daftar lengkap fungsi bawaan modul mysql sesuai binding C terdaftar:

  • mysql.connect(host, user, pass, db, port) — Membuka koneksi baru.
  • mysql.connect_options(config_object) — Membuka koneksi berbasis konfigurasi objek.
  • mysql.query(db, sql) — Menjalankan query SQL mentah.
  • mysql.fetch(db, sql) — Mengambil seluruh baris data sebagai Array of Object.
  • mysql.fetch_one(db, sql) — Mengambil 1 baris pertama hasil query sebagai Object.
  • mysql.error(db) — Mengembalikan pesan kesalahan SQL terakhir.
  • mysql.errno(db) — Mengembalikan kode nomor kesalahan SQL terakhir.
  • mysql.last_id(db) — Mengembalikan ID auto-increment dari insert terakhir.
  • mysql.affected_rows(db) — Mengembalikan jumlah baris yang terpengaruh.
  • mysql.ping(db) — Memeriksa status koneksi server.
  • mysql.select_db(db, db_name) — Mengubah database aktif.
  • mysql.close(db) — Memutuskan dan menutup koneksi.
  • mysql.create_pool(config) — Membuat pool koneksi database.
  • mysql.get_connection(pool) — Mengambil 1 koneksi aktif dari pool.
  • mysql.release_connection(pool, db) — Mengembalikan koneksi ke dalam pool.

🔌 Function mysql.connect()

PAHAT
// Sintaks: mysql.connect(host, user, pass, db_name, port) -> Integer (Pointer) / NULL
db = mysql.connect("127.0.0.1", "root", "secret", "my_db", 3306);

🔍 Function mysql.query()

PAHAT
// Sintaks: mysql.query(handler, sql_string) -> Boolean
berhasil = mysql.query(db, "UPDATE produk SET stok = 50 WHERE id = 12");

📦 Function mysql.fetch() & mysql.fetch_one()

mysql.fetch() mengambil seluruh baris data sebagai array objek, sedangkan mysql.fetch_one() mengambil 1 baris pertama sebagai objek tunggal.

🔢 Function mysql.last_id()

Mengembalikan ID bertipe integer yang dihasilkan oleh kolom `AUTO_INCREMENT` pada operasi `INSERT` terakhir.

✅ Function mysql.affected_rows()

Mengembalikan jumlah baris yang termodifikasi oleh perintah `INSERT`, `UPDATE`, atau `DELETE` terakhir.

// Atau jika gagal:

🧪 Skrip Studi Kasus CRUD Lengkap

PAHAT
impor "mysql";

// 1. Inisialisasi Koneksi
db = mysql.connect("localhost", "root", "", "pahat_db", 3306);

jika (!db) {
    cetak("Koneksi gagal: " + mysql.error());
    os.exit(1);
}

cetak("=== APP PENGELOLAAN INVENTARIS PAHAT ===");

// 2. Buat Tabel Produk
mysql.query(db, "CREATE TABLE IF NOT EXISTS barang (id INT AUTO_INCREMENT PRIMARY KEY, nama VARCHAR(50), harga INT)");

// 3. Tambah Barang Baru (Create)
sql_add = "INSERT INTO barang (nama, harga) VALUES ('Laptop PAHAT Pro', 15000000)";
mysql.query(db, sql_add);
id_barang = mysql.last_id(db);
cetak("Barang ditambahkan dengan ID: " + id_barang);

// 4. Baca Semua Barang (Read)
cetak("--- DAFTAR BARANG ---");
items = mysql.fetch(db, "SELECT * FROM barang");
ulang (i = 0; i < panjang(items); i++) {
    item = items[i];
    cetak("#" + item.id + " " + item.nama + " - Rp" + item.harga);
}

// 5. Perbarui Harga Barang (Update)
mysql.query(db, "UPDATE barang SET harga = 14000000 WHERE id = " + id_barang);
cetak("Harga barang ID " + id_barang + " berhasil diperbarui!");

// 6. Tutup Koneksi
mysql.close(db);
cetak("Koneksi ke MySQL ditutup secara aman.");
